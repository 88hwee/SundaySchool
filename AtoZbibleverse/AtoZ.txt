        let currentIndex = 0;
        let currentWord = "";
        let currentLetter = "";

        let guessedLetters = new Set();

        // Visual Effects State
        const canvas = document.getElementById('animationCanvas');
        const ctx = canvas.getContext('2d');
        let elements = [];
        let ambientStars = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            initializeAmbientStars();
        }
        window.addEventListener('resize', resizeCanvas);

        // Constant background stars
        function initializeAmbientStars() {
            ambientStars = [];
            for (let i = 0; i < 35; i++) {
                ambientStars.push({
                    x: Math.random() * canvas.width,
                    y: Math.random() * canvas.height,
                    size: Math.random() * 2 + 1,
                    alpha: Math.random(),
                    speed: Math.random() * 0.015 + 0.005
                });
            }
        }

        // Initialize Core Game logic
        function initWord() {
            const data = vocabulary[currentIndex];
            currentWord = data.word.toUpperCase();
            currentLetter = data.letter;
            currentVerse = data.verse;
            currentReference = data.reference;
            currentVerseZh = data.verseZh;
            currentReferenceZh = data.referenceZh;
            guessedLetters.clear();
            lives = maxLives;

            // UI Adjustments
            document.getElementById('hintLetter').innerText = currentLetter;
            document.getElementById('progressText').innerText = `Word ${currentIndex + 1} of ${vocabulary.length}`;
            document.getElementById('nextBtn').style.display = 'none';
            
            // Show Hangman and Hide Scripture Verse Card during active gameplay
            document.getElementById('hangmanBox').style.display = 'flex';
            document.getElementById('verseCard').style.display = 'none';

            // Clear and completely collapse banner spacing when game is running
            const banner = document.getElementById('bannerMessage');
            banner.style.display = "none";
            banner.className = "banner-message";
            banner.innerText = "";

            renderHangmanVisuals();
            renderWordBlanks();
            renderKeyboard();
        }

        function renderHangmanVisuals() {
            const hangmanPartIds = [
                'part-head',
                'part-torso',
                'part-left-arm',
                'part-right-arm',
                'part-left-leg',
                'part-right-leg'
            ];

            const wrongGuessCount = maxLives - lives;

            hangmanPartIds.forEach((id, index) => {
                const part = document.getElementById(id);
                if (part) {
                    if (index < wrongGuessCount) {
                        part.classList.add('visible');
                    } else {
                        part.classList.remove('visible');
                    }
                }
            });
        }

        function renderWordBlanks() {
            const container = document.getElementById('wordDisplay');
            container.innerHTML = '';

            currentWord.split('').forEach(char => {
                const element = document.createElement('span');
                if (char === ' ') {
                    element.className = 'word-space';
                    element.innerHTML = '&nbsp;';
                } else {
                    element.className = 'word-letter';
                    if (guessedLetters.has(char)) {
                        element.innerText = char;
                        element.classList.add('correct');
                    } else {
                        element.innerText = '';
                    }
                }
                container.appendChild(element);
            });
        }

        function renderKeyboard() {
            const container = document.getElementById('keyboard');
            container.innerHTML = '';
            const alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';

            alphabet.split('').forEach(char => {
                const btn = document.createElement('button');
                btn.className = 'key';
                btn.innerText = char;
                btn.id = `key-${char}`;
                btn.onclick = () => makeGuess(char);
                container.appendChild(btn);
            });
        }

        function makeGuess(letter) {
            if (guessedLetters.has(letter) || lives <= 0) return;

            guessedLetters.add(letter);
            const keyBtn = document.getElementById(`key-${letter}`);

            if (currentWord.includes(letter)) {
                keyBtn.classList.add('correct-guess');
                keyBtn.disabled = true;
                renderWordBlanks();
                checkWinCondition();
            } else {
                keyBtn.classList.add('wrong-guess');
                keyBtn.disabled = true;
                lives--;
                renderHangmanVisuals();
                triggerWrongSparkle(keyBtn);
                if (lives <= 0) {
                    revealLoss();
                }
            }
        }

        function triggerWrongSparkle(buttonElement) {
            const rect = buttonElement.getBoundingClientRect();
            const x = rect.left + rect.width/2;
            const y = rect.top + rect.height/2;
            for(let i=0; i<6; i++){
                const angle = Math.random() * Math.PI * 2;
                const speed = Math.random() * 2 + 1;
                elements.push({
                    type: 'sparkle',
                    x: x,
                    y: y,
                    vx: Math.cos(angle) * speed,
                    vy: Math.sin(angle) * speed,
                    alpha: 1,
                    color: '#e74c3c',
                    size: Math.random() * 2 + 2
                });
            }
        }

        function showScriptureCard() {
            // Hide hangman and reveal the styled bilingual Scripture Memorization Card
            document.getElementById('hangmanBox').style.display = 'none';
            
            const verseCard = document.getElementById('verseCard');
            document.getElementById('verseReference').innerText = `${currentReference} | ${currentReferenceZh}`;
            document.getElementById('verseText').innerText = `"${currentVerse}"`;
            document.getElementById('verseTextZh').innerText = `"${currentVerseZh}"`;
            verseCard.style.display = 'flex';
        }

        function checkWinCondition() {
            const cleanWordLetters = currentWord.replace(/\s/g, '').split('');
            const solved = cleanWordLetters.every(l => guessedLetters.has(l));

            if (solved) {
                document.querySelectorAll('.key').forEach(btn => btn.disabled = true);

                const randQuote = positiveQuotes[Math.floor(Math.random() * positiveQuotes.length)];
                const banner = document.getElementById('bannerMessage');
                banner.innerText = `✨ ${randQuote}\n"${currentLetter} is for ${currentWord}!"`;
                banner.className = "banner-message show win";
                banner.style.display = "flex"; // Make visible when game ends

                // Show the beautiful verses
                showScriptureCard();

                triggerCelebrationFX();
                document.getElementById('nextBtn').style.display = 'inline-flex';
            }
        }

        function revealLoss() {
            document.querySelectorAll('.key').forEach(btn => btn.disabled = true);

            const container = document.getElementById('wordDisplay');
            container.innerHTML = '';
            currentWord.split('').forEach(char => {
                const element = document.createElement('span');
                if (char === ' ') {
                    element.className = 'word-space';
                    element.innerHTML = '&nbsp;';
                } else {
                    element.className = 'word-letter';
                    element.innerText = char;
                    element.style.color = 'var(--error)';
                    element.style.borderBottomColor = 'var(--error)';
                }
                container.appendChild(element);
            });

            const banner = document.getElementById('bannerMessage');
            banner.innerText = `Keep shining! The word was "${currentWord}".\nLet's learn the verse together!`;
            banner.className = "banner-message show lose";
            banner.style.display = "flex"; // Make visible when game ends

            // Show the beautiful verses even on loss to facilitate learning
            showScriptureCard();

            document.getElementById('nextBtn').style.display = 'inline-flex';
        }

        function advanceGame() {
            currentIndex++;
            if (currentIndex >= vocabulary.length) {
                showVictoryModal();
                triggerGrandFinaleFX();
            } else {
                initWord();
            }
        }

        function showVictoryModal() {
            document.getElementById('victoryModal').style.display = 'flex';
        }

        function closeVictoryModal() {
            document.getElementById('victoryModal').style.display = 'none';
            restartEntireGame();
        }

        function restartEntireGame() {
            currentIndex = 0;
            initWord();
        }

        // --- VISUAL EFFECTS ENGINE ---

        function triggerCelebrationFX() {
            // Cap active particles to prevent mobile lag
            if (elements.length > 150) elements = [];

            // 1. Instant glittering 4-pointed stars burst from the solved word location (center of screen)
            for (let i = 0; i < 25; i++) {
                const angle = Math.random() * Math.PI * 2;
                const speed = Math.random() * 3.5 + 1.5;
                elements.push({
                    type: 'glitter-star',
                    x: canvas.width / 2,
                    y: canvas.height * 0.45,
                    vx: Math.cos(angle) * speed,
                    vy: Math.sin(angle) * speed,
                    size: Math.random() * 6 + 5,
                    alpha: 1.0,
                    spin: (Math.random() - 0.5) * 0.08,
                    angle: Math.random() * Math.PI,
                    color: i % 2 === 0 ? '#ffd700' : '#ffffff' // Alternate gold and silver/white
                });
            }

            // 2. Instant Firework shell detonations (2 immediate colorful bursts in the foreground!)
            explodeRocket(canvas.width * 0.35, canvas.height * 0.35, `hsl(${Math.random() * 360}, 100%, 70%)`);
            explodeRocket(canvas.width * 0.65, canvas.height * 0.35, `hsl(${Math.random() * 360}, 100%, 70%)`);

            // 3. Delayed rocket launches from the bottom
            for (let i = 0; i < 2; i++) {
                setTimeout(() => {
                    launchRocket();
                }, i * 350 + 200);
            }

            createAngel();
        }

        function triggerGrandFinaleFX() {
            elements = [];
            for (let i = 0; i < 8; i++) {
                setTimeout(() => {
                    launchRocket();
                    createAngel();
                }, i * 250);
            }
        }

        function launchRocket() {
            if (elements.length > 120) return; // Prevent performance choke on older devices
            const startX = Math.random() * (canvas.width - 200) + 100;
            const targetY = Math.random() * (canvas.height * 0.3) + 80;
            elements.push({
                type: 'rocket',
                x: startX,
                y: canvas.height,
                targetY: targetY,
                vy: - (Math.random() * 3 + 6),
                color: `hsl(${Math.random() * 360}, 100%, 65%)`
            });
        }

        function explodeRocket(x, y, color) {
            for (let i = 0; i < 30; i++) {
                const angle = Math.random() * Math.PI * 2;
                const speed = Math.random() * 3.5 + 1.2;
                elements.push({
                    type: 'particle',
                    x: x,
                    y: y,
                    vx: Math.cos(angle) * speed,
                    vy: Math.sin(angle) * speed,
                    alpha: 1,
                    decay: Math.random() * 0.015 + 0.015,
                    color: color
                });
            }
        }

        function createAngel() {
            const goRight = Math.random() > 0.5;
            elements.push({
                type: 'angel',
                x: goRight ? -100 : canvas.width + 100,
                y: Math.random() * (canvas.height * 0.25) + 60,
                vx: goRight ? (Math.random() * 1.0 + 1.0) : -(Math.random() * 1.0 + 1.0),
                phase: Math.random() * 100,
                scale: Math.random() * 0.12 + 0.52
            });
        }

        function drawAngel(ctx, x, y, scale, phase) {
            ctx.save();
            ctx.translate(x, y);
            ctx.scale(scale, scale);

            // Glowing golden halo
            ctx.beginPath();
            ctx.ellipse(0, -52, 18, 6, 0, 0, Math.PI * 2);
            ctx.strokeStyle = '#ffa500';
            ctx.lineWidth = 4;
            ctx.stroke();

            ctx.beginPath();
            ctx.ellipse(0, -52, 18, 6, 0, 0, Math.PI * 2);
            ctx.strokeStyle = '#ffd700';
            ctx.lineWidth = 2;
            ctx.stroke();

            const wingOffset = Math.sin(phase) * 12;
            
            // Higher opacity wings with bold sky-blue borders
            ctx.fillStyle = 'rgba(215, 240, 255, 0.95)';
            ctx.strokeStyle = '#4a90e2';
            ctx.lineWidth = 3;

            // Left Wing
            ctx.beginPath();
            ctx.moveTo(-10, -20);
            ctx.bezierCurveTo(-55, -55 + wingOffset, -65, -10 + wingOffset, -10, 0);
            ctx.fill();
            ctx.stroke();

            // Right Wing
            ctx.beginPath();
            ctx.moveTo(10, -20);
            ctx.bezierCurveTo(55, -55 + wingOffset, 65, -10 + wingOffset, 10, 0);
            ctx.fill();
            ctx.stroke();

            // Beautiful White Robe with a distinct border outline
            ctx.fillStyle = '#ffffff';
            ctx.strokeStyle = '#b2c2d4';
            ctx.lineWidth = 2.5;
            ctx.beginPath();
            ctx.moveTo(0, -20);
            ctx.lineTo(-24, 35);
            ctx.lineTo(24, 35);
            ctx.closePath();
            ctx.fill();
            ctx.stroke();

            // Sashes/Ribbon
            ctx.fillStyle = '#2980b9';
            ctx.beginPath();
            ctx.moveTo(-8, 5);
            ctx.lineTo(-20, 25);
            ctx.lineTo(-12, 28);
            ctx.closePath();
            ctx.fill();

            ctx.beginPath();
            ctx.moveTo(8, 5);
            ctx.lineTo(20, 25);
            ctx.lineTo(12, 28);
            ctx.closePath();
            ctx.fill();

            // Head / Skin
            ctx.fillStyle = '#ffe0bd';
            ctx.beginPath();
            ctx.arc(0, -26, 12, 0, Math.PI * 2);
            ctx.fill();
            ctx.strokeStyle = '#e0b388';
            ctx.lineWidth = 1.5;
            ctx.stroke();

            // Friendly cartoon eyes and smile (Clear & high contrast)
            ctx.strokeStyle = '#2c3e50';
            ctx.lineWidth = 2;
            
            // Left eye (happy curved)
            ctx.beginPath();
            ctx.arc(-4, -27, 2, 0, Math.PI, true);
            ctx.stroke();

            // Right eye (happy curved)
            ctx.beginPath();
            ctx.arc(4, -27, 2, 0, Math.PI, true);
            ctx.stroke();

            // Sweet happy smile
            ctx.strokeStyle = '#e74c3c';
            ctx.beginPath();
            ctx.arc(0, -23, 3.5, 0, Math.PI);
            ctx.stroke();

            ctx.restore();
        }

        function updateAndDrawFX() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Draw constant ambient background stars
            for (let i = 0; i < ambientStars.length; i++) {
                const star = ambientStars[i];
                star.alpha += star.speed;
                if (star.alpha > 1 || star.alpha < 0) {
                    star.speed = -star.speed;
                }
                ctx.save();
                ctx.globalAlpha = Math.max(0, Math.min(1, star.alpha));
                ctx.fillStyle = '#fff69b';
                ctx.beginPath();
                ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }

            for (let i = elements.length - 1; i >= 0; i--) {
                const el = elements[i];

                if (el.type === 'sparkle') {
                    el.x += el.vx;
                    el.y += el.vy;
                    el.alpha -= 0.03;
                    ctx.save();
                    ctx.globalAlpha = Math.max(0, el.alpha);
                    ctx.fillStyle = el.color;
                    ctx.beginPath();
                    ctx.arc(el.x, el.y, el.size, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.restore();
                    if (el.alpha <= 0) elements.splice(i, 1);
                }

                else if (el.type === 'glitter-star') {
                    el.x += el.vx;
                    el.y += el.vy;
                    el.vy += 0.04; // Gentle gravity drift
                    el.angle += el.spin;
                    el.alpha -= 0.012; // Slow fade for a glittering trail
                    
                    ctx.save();
                    ctx.globalAlpha = Math.max(0, el.alpha);
                    ctx.fillStyle = el.color;
                    ctx.shadowColor = el.color;
                    ctx.shadowBlur = 8;
                    ctx.translate(el.x, el.y);
                    ctx.rotate(el.angle);
                    
                    // Draw a gorgeous 4-point glittering star
                    ctx.beginPath();
                    for (let s = 0; s < 4; s++) {
                        ctx.lineTo(0, -el.size);
                        ctx.rotate(Math.PI / 4);
                        ctx.lineTo(0, -el.size / 3.5);
                        ctx.rotate(Math.PI / 4);
                    }
                    ctx.closePath();
                    ctx.fill();
                    ctx.restore();

                    if (el.alpha <= 0) elements.splice(i, 1);
                }
                
                else if (el.type === 'rocket') {
                    el.y += el.vy;
                    // Spark trail optimized with mathematical skip (only render spark every other frame)
                    if (Math.random() < 0.5) {
                        elements.push({
                            type: 'sparkle',
                            x: el.x,
                            y: el.y,
                            vx: (Math.random() - 0.5) * 1.5,
                            vy: 1,
                            alpha: 0.8,
                            color: '#f39c12',
                            size: Math.random() * 2 + 1
                        });
                    }

                    if (el.y <= el.targetY || el.vy >= 0) {
                        explodeRocket(el.x, el.y, el.color);
                        elements.splice(i, 1);
                    }
                }
                
                else if (el.type === 'particle') {
                    el.x += el.vx;
                    el.y += el.vy;
                    el.vy += 0.08;
                    el.alpha -= el.decay;

                    ctx.save();
                    ctx.globalAlpha = Math.max(0, el.alpha);
                    ctx.fillStyle = el.color;
                    ctx.beginPath();
                    ctx.arc(el.x, el.y, 3.5, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.restore();

                    if (el.alpha <= 0) elements.splice(i, 1);
                }
                
                else if (el.type === 'angel') {
                    el.x += el.vx;
                    el.phase += 1;
                    
                    drawAngel(ctx, el.x, el.y, el.scale, el.phase * 0.12);

                    // Drop twinkling star trails occasionally (throttled to save CPU)
                    if (Math.random() < 0.1) {
                        elements.push({
                            type: 'glitter-star',
                            x: el.x + (Math.random() - 0.5) * 20,
                            y: el.y + 15,
                            vx: (Math.random() - 0.5) * 1,
                            vy: Math.random() * 1 + 0.5,
                            size: Math.random() * 4 + 3,
                            alpha: 0.8,
                            spin: (Math.random() - 0.5) * 0.08,
                            angle: Math.random() * Math.PI,
                            color: '#ffd700'
                        });
                    }

                    if ((el.vx > 0 && el.x > canvas.width + 150) || (el.vx < 0 && el.x < -150)) {
                        elements.splice(i, 1);
                    }
                }
            }

            requestAnimationFrame(updateAndDrawFX);
        }

        // Setup Init
        window.onload = function () {
            resizeCanvas();
            initWord();
            updateAndDrawFX();
        };
